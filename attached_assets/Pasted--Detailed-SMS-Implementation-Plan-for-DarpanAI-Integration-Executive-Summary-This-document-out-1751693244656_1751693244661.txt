# Detailed SMS Implementation Plan for DarpanAI Integration

## Executive Summary

This document outlines the comprehensive implementation plan for integrating the Advanced Scholarship Matching System (SMS) into the existing DarpanAI platform. The integration leverages the existing authentication, database infrastructure, and admin system while adding specialized scholarship matching capabilities.

## 1. Current System Analysis

### 1.1 Existing DarpanAI Architecture
- **Frontend**: React 18 with TypeScript, Wouter routing, TanStack Query, shadcn/ui components
- **Backend**: Express.js, Passport.js authentication, Drizzle ORM
- **Database**: PostgreSQL with existing tables (users, analyses, appointments)
- **Authentication**: Role-based access (admin/user) with session management
- **Current Features**: Visa rejection analysis, consultation booking, quota management

### 1.2 Integration Approach
The SMS will be integrated as a new module within the existing DarpanAI platform, maintaining consistency with current architecture and design patterns.

### 3.2 Matching Algorithm Implementation

```typescript
// server/services/scholarshipMatching.ts
class ScholarshipMatcher {
  async calculateMatchScore(
    scholarship: Scholarship,
    userPreferences: UserPreferences
  ): Promise<{ score: number; reasons: string[] }> {
    let score = 0;
    const reasons: string[] = [];

    // Major/Field matching (30 points)
    if (this.matchesMajor(scholarship.tags, userPreferences.major)) {
      score += 30;
      reasons.push('Matches your major');
    }

    // GPA requirements (25 points)
    if (this.meetsGPARequirement(scholarship, userPreferences.gpa)) {
      score += 25;
      reasons.push('Meets GPA requirement');
    }

    // Financial need (20 points)
    if (scholarship.need_based && userPreferences.financial_need) {
      score += 20;
      reasons.push('Matches financial need');
    }

    // Level of study (15 points)
    if (scholarship.level_of_study.includes(userPreferences.level_of_study)) {
      score += 15;
      reasons.push('Matches your level of study');
    }

    // Demographic matching (10 points)
    if (this.matchesDemographics(scholarship, userPreferences)) {
      score += 10;
      reasons.push('Matches your profile');
    }

    return { score: Math.min(score, 100), reasons };
  }

  private matchesMajor(scholarshipTags: string[], userMajor: string): boolean {
    // Implementation for major matching logic
    return scholarshipTags.some(tag => 
      tag.toLowerCase().includes(userMajor.toLowerCase())
    );
  }

  // Additional matching methods...
}
```

### 7.2 Caching Strategy

```typescript
// server/services/cacheService.ts
import NodeCache from 'node-cache';

class CacheService {
  private cache = new NodeCache({ stdTTL: 600 }); // 10 minutes

  async getScholarships(filters: ScholarshipFilters): Promise<Scholarship[]> {
    const cacheKey = `scholarships:${JSON.stringify(filters)}`;
    
    let scholarships = this.cache.get<Scholarship[]>(cacheKey);
    if (!scholarships) {
      scholarships = await this.fetchScholarshipsFromDB(filters);
      this.cache.set(cacheKey, scholarships);
    }
    
    return scholarships;
  }

  invalidateScholarshipCache() {
    this.cache.flushAll();
  }
}
```

```typescript
// server/services/recommendationEngine.ts
class RecommendationEngine {
  async generatePersonalizedRecommendations(userId: number): Promise<Scholarship[]> {
    const userPreferences = await this.getUserPreferences(userId);
    const userHistory = await this.getUserScholarshipHistory(userId);
    const similarUsers = await this.findSimilarUsers(userId);
    
    // Collaborative filtering
    const collaborativeRecommendations = await this.collaborativeFiltering(
      userId, 
      similarUsers
    );
    
    // Content-based filtering
    const contentBasedRecommendations = await this.contentBasedFiltering(
      userPreferences
    );
    
    // Hybrid approach: combine both methods
    return this.mergeRecommendations(
      collaborativeRecommendations,
      contentBasedRecommendations
    );
  }

  private async collaborativeFiltering(
    userId: number, 
    similarUsers: number[]
  ): Promise<Scholarship[]> {
    const scholarshipsLikedBySimilarUsers = await db
      .select()
      .from(user_scholarships)
      .where(
        and(
          inArray(user_scholarships.user_id, similarUsers),
          eq(user_scholarships.status, 'saved')
        )
      )
      .groupBy(user_scholarships.scholarship_id)
      .having(count() > 2); // Liked by at least 2 similar users

    return scholarshipsLikedBySimilarUsers;
  }

  private async contentBasedFiltering(
    preferences: UserPreferences
  ): Promise<Scholarship[]> {
    const query = db
      .select()
      .from(scholarships)
      .where(
        and(
          eq(scholarships.is_active, true),
          or(
            arrayContains(scholarships.tags, preferences.preferred_tags),
            arrayContains(scholarships.level_of_study, [preferences.level_of_study])
          )
        )
      );

    return await query.limit(20);
  }
}
```

```typescript
// server/services/notificationService.ts
class NotificationService {
  async scheduleDeadlineNotifications(): Promise<void> {
    const upcomingDeadlines = await db
      .select()
      .from(scholarships)
      .innerJoin(user_scholarships, eq(scholarships.id, user_scholarships.scholarship_id))
      .innerJoin(users, eq(user_scholarships.user_id, users.id))
      .where(
        and(
          between(
            scholarships.deadline,
            new Date(),
            new Date(Date.now() + 30 * 24 * 60 * 60 * 1000) // Next 30 days
          ),
          eq(user_scholarships.status, 'saved')
        )
      );

    for (const deadline of upcomingDeadlines) {
      await this.sendDeadlineNotification(deadline);
    }
  }

  private async sendDeadlineNotification(deadline: any): Promise<void> {
    const daysUntilDeadline = Math.ceil(

    );

    const emailContent = this.generateDeadlineEmailContent(
      deadline.scholarship_name,

      daysUntilDeadline
    );

    await this.sendEmail(deadline.email, emailContent);
```

```typescript
            This scholarship offers {scholarship.amount_display} with a deadline of {formatDate(scholarship.deadline)}.
            Available for {scholarship.level_of_study.join(' and ')} students.
            {scholarship.need_based ? ' This is a need-based scholarship.' : ''}
            {scholarship.merit_based ? ' This is a merit-based scholarship.' : ''}
          </p>
        </div>

        {/* Keyboard navigation support */}
        <div className="flex gap-2 mt-4">
          <Button
            variant="outline"
            size="sm"
            onClick={() => onViewDetails(scholarship)}
            className="flex-1"
            aria-describedby={`scholarship-title-${scholarship.id}`}
          >
            View Details
          </Button>
          <Button
            variant="ghost"
            size="sm"
            onClick={() => onSave(scholarship.id)}
            className="px-3"
            aria-label={`Save ${scholarship.scholarship_name} scholarship`}
          >
            <Heart className="w-4 h-4" />
          </Button>
        </div>
      </CardContent>
    </Card>
  );
};
```

```typescript
// server/middleware/sanitization.ts
import DOMPurify from 'isomorphic-dompurify';
import { body, query } from 'express-validator';

export const sanitizeScholarshipInput = [
  body('scholarship_name').trim().escape().isLength({ min: 1, max: 500 }),
  body('description').customSanitizer(value => DOMPurify.sanitize(value)),
  body('tags.*').trim().escape().isLength({ max: 50 }),
  body('eligibility_summary.*').trim().escape().isLength({ max: 1000 }),
  query('search').optional().trim().escape().isLength({ max: 100 }),
  query('tags').optional().isArray({ max: 10 })
];
```
